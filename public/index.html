require("dotenv").config(); const express = require("express"); const app = express(); app.use(express.json({ limit: "1mb" })); app.use(express.static("public")); // Проверка работы сервера app.get("/health", (req, res) => { res.json({ status: "ONLINE", name: "NEON AI" }); }); // Проверка настроек app.get("/api/config", (req, res) => { res.json({ status: "ONLINE", model: process.env.MODEL || "not configured" }); }); // Основной ИИ-чат app.post("/api/chat", async (req, res) => { try { const messages = req.body.messages; if (!Array.isArray(messages)) { return res.status(400).json({ error: "Неверный формат сообщений" }); } if (!process.env.OPENROUTER_API_KEY) { return res.status(500).json({ error: "API-ключ OpenRouter не найден" }); } const response = await fetch( "https://openrouter.ai/api/v1/chat/completions", { method: "POST", headers: { "Authorization": `Bearer ${process.env.OPENROUTER_API_KEY}`, "Content-Type": "application/json", "HTTP-Referer": "https://neon-ai.onrender.com", "X-Title": "NEON AI" }, body: JSON.stringify({ model: process.env.MODEL || "openrouter/free", messages: [ { role: "system", content: "Ты NEON AI — дружелюбный умный ИИ. Отвечай на русском языке, понятно и полезно." }, ...messages.slice(-20) ] }) } ); const data = await response.json(); if (!response.ok) { console.log("OPENROUTER STATUS:", response.status); console.log("OPENROUTER ERROR:", data); return res.status(502).json({ error: data.error?.message || "Ошибка OpenRouter" }); } const reply = data.choices?.[0]?.message?.content; res.json({ reply: reply || "NEON AI не получил ответ." }); } catch (error) { console.log("SERVER ERROR:", error.message); res.status(500).json({ error: "Ошибка сервера: " + error.message }); } }); // Запуск для Render const PORT = process.env.PORT || 3000; app.listen(PORT, "0.0.0.0", () => { console.log(`NEON AI ONLINE на порту ${PORT}`); });
